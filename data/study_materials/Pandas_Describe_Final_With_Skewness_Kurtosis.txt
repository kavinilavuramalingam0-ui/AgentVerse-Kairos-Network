    "# Import required libraries\n",
    "import pandas as pd\n",
    "import matplotlib.pyplot as plt\n",
    "\n",
    "# Step 1: Create the dataset\n",
    "data = [4120,5230,6780,4520,7890,\n",
    "        6230,5120,9340,7180,6450,\n",
    "        5870,4320,7650,5230,6120,\n",
    "        8420,7320,5230,6540,4780]\n",
    "\n",
    "# Step 2: Convert the list into a Pandas Series\n",
    "s = pd.Series(data)\n",
    "\n",
    "# -----------------------------\n",
    "# Summary Statistics\n",
    "# -----------------------------\n",
    "print(s.describe())\n",
    "\n",
    "print(\"\\nMode                  :\", s.mode().tolist())\n",
    "print(\"Population Variance   :\", s.var(ddof=0))\n",
    "print(\"Sample Variance       :\", s.var(ddof=1))\n",
    "print(\"Skewness              :\", s.skew())\n",
    "print(\"Kurtosis              :\", s.kurt())\n",
    "print(\"Range                 :\", s.max()-s.min())\n",
    "\n",
    "# Frequency Bar Chart\n",
    "freq = s.value_counts().sort_index()\n",
    "plt.figure(figsize=(9,4))\n",
    "plt.bar(freq.index.astype(str), freq.values)\n",
    "plt.title(\"Frequency Bar Chart\")\n",
    "plt.xlabel(\"Data Value\")\n",
    "plt.ylabel(\"Frequency\")\n",
    "plt.xticks(rotation=45)\n",
    "plt.show()\n",
    "\n",
    "# Histogram\n",
    "plt.figure(figsize=(8,4))\n",
    "plt.hist(s,bins=6,edgecolor=\"black\")\n",
    "plt.title(\"Histogram\")\n",
    "plt.xlabel(\"Value\")\n",
    "plt.ylabel(\"Frequency\")\n",
    "plt.show()\n",
    "\n",
    "# Pie Chart\n",
    "plt.figure(figsize=(6,6))\n",
    "plt.pie(freq,\n",
    "        labels=freq.index,\n",
    "        autopct=\"%1.1f%%\",\n",
    "        startangle=90)\n",
    "plt.title(\"Pie Chart of Data Values\")\n",
